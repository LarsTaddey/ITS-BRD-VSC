46static void print_values(uint8_t x_coordinate, uint8_t y_coordinate,
47 double value, char *previous_value, int index) {
48 char text[TXT_LCD_LENGTH + 1];
49
50 snprintf(text, sizeof(text), "%10.1f", value);
51 if (previous_value[index] != text[index]) {
52 lcdGotoXY(x_coordinate + index, y_coordinate);
53 lcdPrintC(text[index]);
54 previous_value[index] = text[index];
55 }
56}

125void output_data_to_lcd(const AngleData *angle_data) {
126 // Needed to be able to compare previous and current values
127 static char previous_angle[TXT_LCD_LENGTH];
128 static char previous_speed[TXT_LCD_LENGTH];
129 static uint8_t index = 0;
130 static bool first = true;
131
132 // Only update one of the two values at once. If first one is done then do
133 // the second one
134 if (first) {
135 print_values(X_COORDINATE + X_OFFSET, ANGLE_Y, angle_data->angle,
136 previous_angle, index);
137 } else {
138
139 print_values(X_COORDINATE + X_OFFSET, SPEED_Y, angle_data->angle_speed,
140 previous_speed, index);
141 }
142
143 ++index;
144 if (index >= TXT_LCD_LENGTH) {
145 first = !first;
146 index = 0;
147 }
148}
