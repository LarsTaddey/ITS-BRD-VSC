27int main(void) {
28 PhaseData previous_phase;

30 PhaseData reference_phase;

35 // Initialization on start

37 initTimer();
38 output_init();
39 state = STATE_INIT;
40
41 while (1) {
42 if (clear_button_pressed()) {
43 state = STATE_INIT;
44 }
45 // FSM with three states
46 switch (state) {
47 // Define variables and set default output
48 case STATE_INIT:
49
50 set_default_output();
51 previous_phase.phase_counter = 0;
52 previous_phase.timestamp = 0;
53 current_phase = previous_phase;
54 reference_phase = previous_phase;
55 read_pins_and_time(&previous_phase);
56 state = STATE_NORMAL;
57 break;
58
59 // Read pins and check if a phase change occured, then calculate angle and
60 // angle speed
61 case STATE_NORMAL:
62
63 read_pins_and_time(&current_phase);
64 direction = phase_change(&previous_phase, &current_phase);
65
66 if ((current_phase.timestamp - reference_phase.timestamp) >
67 REFRESH_DELAY) {
68 angle_data.angle = calculate_angle(&current_phase);
69 angle_data.angle_speed =
70 calculate_angular_speed(&reference_phase, &current_phase);
71
72 reference_phase = current_phase;
73 }
74
75 previous_phase = current_phase;
76
77 set_counter_leds(current_phase.phase_counter);
78 output_data_to_lcd(&angle_data);
79
80 if (direction == DIRECTION_FORWARD) {
81 turn_off_backward_led();
82 turn_on_forward_led();
83 }
84 if (direction == DIRECTION_BACKWARD) {
85 turn_off_forward_led();
86 turn_on_backward_led();
87 }
88 if (direction == INVALID_TRANSITION) {
89 turn_on_error_led();
90 state = STATE_ERROR;
91 }
92 break;
93
94 // Waits for clear_button to be pressed and then resets the state
95 case STATE_ERROR:
96 if (clear_button_pressed()) {
97 state = STATE_INIT;
98 }
99 break;
100
101 default:
102 state = STATE_INIT;
103 break;
104 }
105 }
106}
